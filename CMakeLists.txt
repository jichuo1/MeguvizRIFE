cmake_minimum_required(VERSION 3.31)
project(MeguvizRIFE VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
if(NOT TARGET MeguvizCompute::backend_ncnn_sdk)
  find_package(MeguvizCompute 0.1 CONFIG REQUIRED)
endif()
get_target_property(MGR_PRIVATE_FP MeguvizCompute::backend_ncnn_sdk INTERFACE_MGC_PRIVATE_SDK_FINGERPRINT)
if(NOT MGR_PRIVATE_FP MATCHES "^[0-9a-f]+$" OR NOT MGR_PRIVATE_FP)
  message(FATAL_ERROR "The Compute SDK must expose its exact private fingerprint")
endif()
file(GLOB_RECURSE identity_sources CONFIGURE_DEPENDS include/*.h src/*.cpp src/*.h src/*.hpp)
list(APPEND identity_sources "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt")
list(SORT identity_sources)
set(identity_text "sdk=${MGR_PRIVATE_FP};compiler=${CMAKE_CXX_COMPILER_ID}-${CMAKE_CXX_COMPILER_VERSION};system=${CMAKE_SYSTEM_NAME};abi=${ANDROID_ABI};config=${CMAKE_BUILD_TYPE};flags=${CMAKE_CXX_FLAGS};release=${CMAKE_CXX_FLAGS_RELEASE};stl=${CMAKE_MSVC_RUNTIME_LIBRARY}-${ANDROID_STL}")
foreach(file IN LISTS identity_sources)
  file(SHA256 "${file}" content_hash)
  file(RELATIVE_PATH logical_path "${CMAKE_CURRENT_SOURCE_DIR}" "${file}")
  string(APPEND identity_text ";${logical_path}=${content_hash}")
endforeach()
string(SHA256 MGR_BUILD_FP "${identity_text}")
configure_file(cmake/build_identity.hpp.in generated/build_identity.hpp @ONLY)
configure_file(cmake/build-info.json.in build-info.json @ONLY)
set(legacy src/backends/ncnn/legacy)
set(shader_headers)
foreach(shader upstream/warp upstream/warp_pack4 frame_preproc frame_postproc frame_fused frame_fused_rgb8 frame_reduce)
  get_filename_component(name "${shader}" NAME)
  set(header "${CMAKE_CURRENT_BINARY_DIR}/generated/${name}.comp.hex.h")
  add_custom_command(OUTPUT "${header}" COMMAND "${CMAKE_COMMAND}"
    "-DSOURCE=${CMAKE_CURRENT_SOURCE_DIR}/${legacy}/${shader}.comp" "-DOUTPUT=${header}"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/${legacy}/embed_shader.cmake"
    DEPENDS "${legacy}/${shader}.comp" "${legacy}/embed_shader.cmake" VERBATIM)
  list(APPEND shader_headers "${header}")
endforeach()
add_library(meguviz_rife STATIC src/core/rife.cpp "${legacy}/frame_engine.cpp" "${legacy}/upstream/rife_warp.cpp" ${shader_headers})
add_library(MeguvizRIFE::core ALIAS meguviz_rife)
set_target_properties(meguviz_rife PROPERTIES EXPORT_NAME core CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
target_include_directories(meguviz_rife PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
  PRIVATE src "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(meguviz_rife PRIVATE MeguvizCompute::backend_ncnn_sdk PUBLIC MeguvizCompute::core)
if(MSVC)
  target_compile_options(meguviz_rife PRIVATE /EHsc /W4)
else()
  set_source_files_properties(src/core/rife.cpp "${legacy}/frame_engine.cpp" PROPERTIES COMPILE_OPTIONS -fexceptions)
  target_compile_options(meguviz_rife PRIVATE -Wall -Wextra)
endif()
add_library(meguviz_rife_ncnn INTERFACE)
add_library(MeguvizRIFE::ncnn ALIAS meguviz_rife_ncnn)
set_target_properties(meguviz_rife_ncnn PROPERTIES EXPORT_NAME ncnn)
target_link_libraries(meguviz_rife_ncnn INTERFACE meguviz_rife)
option(MGR_BUILD_CLI "Build the independent two-image reference CLI" ON)
set(MGR_REQUIRED_PRIVATE_SDK_FINGERPRINT "" CACHE STRING "Required Compute private SDK fingerprint")
if(MGR_REQUIRED_PRIVATE_SDK_FINGERPRINT)
  get_target_property(sdk_identity MeguvizCompute::backend_ncnn_sdk INTERFACE_MGC_PRIVATE_SDK_FINGERPRINT)
  if(NOT sdk_identity STREQUAL MGR_REQUIRED_PRIVATE_SDK_FINGERPRINT)
    message(FATAL_ERROR "RIFE private SDK fingerprint mismatch")
  endif()
endif()
if(MGR_BUILD_CLI)
  add_executable(mgr-cli src/cli/main.cpp)
  target_include_directories(mgr-cli PRIVATE src)
  target_include_directories(mgr-cli PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_link_libraries(mgr-cli PRIVATE meguviz_rife MeguvizCompute::backend_ncnn_sdk)
  if(MSVC)
    target_compile_options(mgr-cli PRIVATE /EHsc)
  endif()
  install(TARGETS mgr-cli RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
install(TARGETS meguviz_rife meguviz_rife_ncnn EXPORT MeguvizRIFETargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES models/catalog/rife46.manifest.json LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/MeguvizRIFE)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/build-info.json" DESTINATION ${CMAKE_INSTALL_DATADIR}/MeguvizRIFE)
install(FILES "${legacy}/NOTICE.md" "${legacy}/upstream/LICENSE" DESTINATION ${CMAKE_INSTALL_DATADIR}/MeguvizRIFE/notices)
configure_package_config_file(cmake/MeguvizRIFEConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/MeguvizRIFEConfig.cmake" INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MeguvizRIFE)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/MeguvizRIFEConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(EXPORT MeguvizRIFETargets NAMESPACE MeguvizRIFE:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MeguvizRIFE)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/MeguvizRIFEConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/MeguvizRIFEConfigVersion.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MeguvizRIFE)
option(MGR_BUILD_TESTS "Build independent C ABI boundary checks" ON)
if(MGR_BUILD_TESTS)
  enable_testing()
  add_executable(mgr_c_consumer tests/c_consumer.c)
  target_link_libraries(mgr_c_consumer PRIVATE MeguvizRIFE::core)
  set_target_properties(mgr_c_consumer PROPERTIES LINKER_LANGUAGE CXX)
  add_test(NAME public_c_model_bounds COMMAND mgr_c_consumer)
  add_executable(mgr-legacy-oracle tests/oracle/main.cpp tests/oracle/frame_engine.cpp tests/oracle/upstream/rife_warp.cpp ${shader_headers})
  target_include_directories(mgr-legacy-oracle PRIVATE tests/oracle "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_link_libraries(mgr-legacy-oracle PRIVATE MeguvizCompute::backend_ncnn_sdk)
  if(MSVC)
    target_compile_options(mgr-legacy-oracle PRIVATE /EHsc)
  else()
    set_source_files_properties(tests/oracle/frame_engine.cpp PROPERTIES COMPILE_OPTIONS -fexceptions)
  endif()
endif()
